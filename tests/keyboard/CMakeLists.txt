cmake_minimum_required(VERSION 3.22)
project(screen_keyboard_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LingmoSDK REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-server)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../src)
add_library(screen-keyboard-test-code STATIC ${SRC}/shell/keyboard/keyboard_model.cpp ${SRC}/shell/keyboard/keyboard_view.cpp ${SRC}/input/xkb_tracker.cpp ${SRC}/core/logging.cpp)
target_include_directories(screen-keyboard-test-code PUBLIC ${SRC})
target_compile_options(screen-keyboard-test-code PUBLIC -include${SRC}/namespace_compat.h)
target_link_libraries(screen-keyboard-test-code PUBLIC Lingmo::SDK PkgConfig::XKB PkgConfig::WAYLAND)
add_executable(screen-keyboard-test keyboard_test.cpp)
target_link_libraries(screen-keyboard-test PRIVATE screen-keyboard-test-code)
enable_testing()
add_test(NAME keyboard-model-ui COMMAND screen-keyboard-test)
set_tests_properties(keyboard-model-ui PROPERTIES PASS_REGULAR_EXPRESSION "Screen keyboard tests passed")
add_executable(screen-keyboard-capture capture.cpp)
target_link_libraries(screen-keyboard-capture PRIVATE screen-keyboard-test-code)
# Compile the real shell controller against compositor headers without requiring a GPU compositor link.
add_library(screen-keyboard-controller-check OBJECT ${SRC}/shell/keyboard/screen_keyboard.cpp)
target_link_libraries(screen-keyboard-controller-check PRIVATE screen-keyboard-test-code)

find_package(GTest REQUIRED)
add_library(screen-keyboard-shell-test-check OBJECT ../shell/screen_keyboard_test.cpp)
target_include_directories(screen-keyboard-shell-test-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(screen-keyboard-shell-test-check PRIVATE screen-keyboard-test-code GTest::gtest)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_executable(screen-keyboard-shortcut-test shortcut_test.cpp ${SRC}/accelerators/accelerator_actions.cpp ${SRC}/accelerators/accelerator_delegate.cpp)
target_link_libraries(screen-keyboard-shortcut-test PRIVATE screen-keyboard-test-code)
add_test(NAME keyboard-shortcut COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:screen-keyboard-shortcut-test>)
set_tests_properties(keyboard-shortcut PROPERTIES PASS_REGULAR_EXPRESSION "Keyboard shortcut contract passed")

add_executable(screen-keyboard-cli-check ${SRC}/shell/keyboard/main.cpp)
target_link_libraries(screen-keyboard-cli-check PRIVATE Lingmo::SDKBase)
add_test(NAME keyboard-cli-help COMMAND screen-keyboard-cli-check --help)
