cmake_minimum_required(VERSION 3.22)
project(shell_appearance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(LingmoSDK REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-server)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
pkg_check_modules(FONTS REQUIRED IMPORTED_TARGET fontconfig freetype2 harfbuzz icu-uc icu-i18n)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../src)
add_library(review-code STATIC ${SRC}/decoration/text.cpp ${SRC}/decoration/raster.cpp ${SRC}/decoration/decoration_theme.cpp ${SRC}/decoration/window_visual_style.cpp ${SRC}/render/rounded_shape.cpp ${SRC}/core/preferences.cpp ${SRC}/shell/wallpaper/bar_backdrop.cpp ${APPEARANCE_SOURCE})
target_include_directories(review-code PUBLIC ${SRC} ${FRAMEWORK_INCLUDE})
target_compile_options(review-code PUBLIC -include${SRC}/namespace_compat.h)
target_link_libraries(review-code PUBLIC Lingmo::SDK PkgConfig::WAYLAND PkgConfig::XKB PkgConfig::FONTS)
add_executable(shell-appearance-review review.cpp)
target_link_libraries(shell-appearance-review PRIVATE review-code)
add_library(shell-host-check OBJECT ${SRC}/shell/shell.cpp ${SRC}/shell/wallpaper/wallpaper.cpp ${SRC}/shell/statusbar/status_bar_window.cpp ${SRC}/shell/statusbar/status_icon.cpp ${SRC}/shell/statusbar/status_bar_backdrop.cpp ${SRC}/shell/dock/dock_window.cpp)
target_link_libraries(shell-host-check PRIVATE review-code)
enable_testing()
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_test(NAME shell-appearance COMMAND ${DBUS_RUN_SESSION} --config-file=${BUS_CONFIG} -- $<TARGET_FILE:shell-appearance-review>)
